/**
 * 🤿 블루독 프리다이빙 수업 예약 시스템 (구글 Apps Script) — 버전 2
 *
 * 하는 일
 *  - 수강생이 날짜를 고르고, 5가지 고정 수업 중 원하는 것을 선택해 예약해요.
 *  - 겹치는 수업(10시 이론 ↔ 북항 1부, 북항 3부 ↔ 7시 이론)은 함께 고를 수 없어요.
 *  - 정원이 차면 자동 마감, 예약은 '예약명단' 시트에 자동 기록돼요.
 *  - 예약될 때마다 구글 캘린더에 수업 일정 + 명단 + 렌탈 장비가 자동 정리돼요.
 *
 * 처음 한 번: 위쪽 함수 선택 칸에서 'setup' 선택 → ▶ 실행
 */

const SHEET_CLASSES = '수업종류';
const SHEET_BOOKINGS = '예약명단';
const SHEET_OFF = '휴무';
const SHEET_SETTINGS = '설정';
const CANCELLED = '취소';
const RENTAL_ITEMS = ['슈트', '핀', '웨이트', '스노클'];
const WEEK = ['일', '월', '화', '수', '목', '금', '토'];
const HOLIDAY_CAL_DEFAULT = 'ko.south_korea#holiday@group.v.calendar.google.com';

/* ───────────── 1. 처음 설치 ───────────── */

function setup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const classes = makeSheet_(ss, SHEET_CLASSES,
    ['수업', '시작', '종료', '정원', '겹치는수업', '운영요일', '장소', '공휴일운영']);
  classes.getRange('B:C').setNumberFormat('@');
  classes.getRange('F:F').setNumberFormat('@');
  if (classes.getLastRow() < 2) {
    // 북항잠수풀은 매주 월요일·공휴일 휴무 → 북항 수업만 '월' 제외·공휴일운영 '아니오' (이론은 매일 가능)
    classes.getRange(2, 1, 5, 8).setValues([
      ['북항 1부', '09:30', '11:30', 4, '10시 이론', '화수목금토일', '북항잠수풀', '아니오'],
      ['10시 이론', '10:00', '12:00', 4, '북항 1부', '월화수목금토일', '블루독 센터', '예'],
      ['북항 2부', '14:00', '16:00', 4, '', '화수목금토일', '북항잠수풀', '아니오'],
      ['북항 3부', '18:10', '20:10', 4, '7시 이론', '화수목금토일', '북항잠수풀', '아니오'],
      ['7시 이론', '19:00', '21:00', 4, '북항 3부', '월화수목금토일', '블루독 센터', '예'],
    ]);
  }

  makeSheet_(ss, SHEET_BOOKINGS,
    ['예약시각', '날짜', '수업', '이름', '연락처', '렌탈장비', '상태']);

  const off = makeSheet_(ss, SHEET_OFF, ['날짜', '수업', '메모']);
  off.getRange('A:A').setNumberFormat('@');

  const settings = makeSheet_(ss, SHEET_SETTINGS, ['항목', '값', '설명']);
  if (settings.getLastRow() < 2) {
    settings.getRange(2, 1, 6, 3).setValues([
      ['예약가능일수', 60, '오늘부터 며칠 뒤까지 예약을 받을지'],
      ['예약마감(시간전)', 12, '수업 시작 몇 시간 전에 예약을 닫을지'],
      ['겹침마감', '예', '예 = 누군가 10시 이론을 예약하면 그날 북항 1부는 모두에게 마감 (강사 1명 기준) / 아니오 = 한 사람이 둘 다 고르는 것만 막음'],
      ['캘린더ID', 'primary', 'primary = 이 구글 계정의 기본 캘린더'],
      ['공휴일캘린더ID', HOLIDAY_CAL_DEFAULT, '구글 "대한민국의 휴일" 캘린더 (대체공휴일 포함, 자동 갱신)'],
      ['관리자비밀번호', '', '관리자 페이지(admin.html) 비밀번호 — 직접 정해서 적어 주세요'],
    ]);
  }

  syncAllCalendar(); // 캘린더 권한을 설치 때 한 번에 받기 위해 실행
  const today = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd');
  const hol = getHolidays_(today, addDays_(today, 90), getSettings_());
  const list = Object.keys(hol).sort().slice(0, 5).map(d => d.slice(5) + ' ' + hol[d]).join(', ');
  SpreadsheetApp.getUi().alert('✅ 설치 완료!\n\n앞으로 90일 공휴일 ' + Object.keys(hol).length + '개를 불러왔어요.\n' +
    (list || '(공휴일을 불러오지 못했어요. 휴무 탭에 직접 적어 주세요)') +
    '\n\n이제 [배포 → 새 배포]를 진행해 주세요.');
}

function makeSheet_(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, headers.length).setValues([headers])
      .setFontWeight('bold').setBackground('#0B2545').setFontColor('#FFFFFF');
    sh.setFrozenRows(1);
  }
  return sh;
}

/* 시트를 열면 위쪽에 '🤿 블루독' 메뉴가 생겨요 */

  SpreadsheetApp.getUi().createMenu('🤿 블루독')
    .addItem('캘린더 다시 맞추기 (취소 반영)', 'syncAllCalendar')
    .addToUi();
}

/* ───────────── 2. 예약 페이지와 대화하는 부분 ───────────── */

// 예약 페이지가 '수업 목록 + 날짜별 빈자리'를 달라고 할 때
function doGet() {
  try {
    return json_(Object.assign({ ok: true }, buildSchedule_()));
  } catch (err) {
    return json_({ ok: false, message: '시간표를 불러오지 못했어요: ' + err.message });
  }
}

// 예약 페이지에서 '예약하기'를 눌렀을 때
function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    const req = JSON.parse(e.postData.contents || '{}');
    if (String(req.action || '').indexOf('admin') === 0) return json_(handleAdmin_(req, lock));

    // ① 입력값 확인
    const name = String(req.name || '').trim();
    const phone = normPhone_(req.phone);
    const date = String(req.date || '').trim();
    const picks = Array.isArray(req.classes) ? req.classes.map(String) : [];
    const rental = normRental_(req.rental);

    if (!name || name.length > 20) return json_({ ok: false, message: '이름을 확인해 주세요.' });
    if (!phone) return json_({ ok: false, message: '연락처를 010-1234-5678 형식으로 입력해 주세요.' });
    if (!picks.length) return json_({ ok: false, message: '수업을 하나 이상 골라 주세요.' });
    if (!rental) return json_({ ok: false, message: '렌탈 장비를 선택해 주세요.' });

    // ② 동시에 여러 명이 눌러도 정원이 넘지 않도록 잠금
    lock.waitLock(20000);

    const sched = buildSchedule_();
    const day = sched.days.find(d => d.date === date);
    if (!day) return json_({ ok: false, message: '이 날짜는 예약할 수 없어요.' });

    const byName = {};
    sched.classes.forEach(c => { byName[c.name] = c; });

    // ③ 고른 수업끼리 겹치는지 확인
    for (const p of picks) {
      if (!byName[p]) return json_({ ok: false, message: '알 수 없는 수업이에요: ' + p });
      const clash = byName[p].conflicts.find(c => picks.indexOf(c) >= 0);
      if (clash) return json_({ ok: false, message: p + '와(과) ' + clash + '은(는) 시간이 겹쳐 함께 예약할 수 없어요.' });
    }

    // ④ 각 수업이 열려 있는지, 같은 사람이 이미 예약했는지 확인
    const existing = activeBookings_();
    for (const p of picks) {
      const slot = day.slots[p];
      if (!slot || slot.closed) {
        const why = slot ? slot.reason : '예약 불가';
        return json_({ ok: false, message: p + '은(는) 지금 예약할 수 없어요 (' + why + '). 다른 수업을 골라 주세요.' });
      }
      if (existing.some(b => b.date === date && b.cls === p && b.phone === phone)) {
        return json_({ ok: false, message: '이미 ' + p + '에 예약되어 있어요.' });
      }
    }

    // ⑤ 예약명단에 기록 (수업마다 한 줄)
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_BOOKINGS);
    const now = new Date();
    picks.forEach(p => sheet.appendRow([now, "'" + date, p, name, "'" + phone, rental, '확정']));
    SpreadsheetApp.flush();

    // ⑥ 캘린더 갱신 (실패해도 예약은 유지)
    try {
      const settings = getSettings_();
      const after = activeBookings_();
      picks.forEach(p => upsertEvent_(date, byName[p], after.filter(b => b.date === date && b.cls === p), settings));
    } catch (calErr) { console.error('캘린더 갱신 실패', calErr); }

    return json_({ ok: true });
  } catch (err) {
    return json_({ ok: false, message: '예약 중 문제가 생겼어요. 잠시 후 다시 시도해 주세요.' });
  } finally {
    try { lock.releaseLock(); } catch (_) {}
  }
}

/* ───────────── 2-1. 관리자 페이지 ───────────── */

function handleAdmin_(req, lock) {
  const settings = getSettings_();
  const cache = CacheService.getScriptCache();

  // 비밀번호를 5번 틀리면 10분 동안 잠겨요 (무작위 대입 방지)
  const fails = Number(cache.get('adminFails') || 0);
  if (fails >= 5) return { ok: false, code: 'locked', message: '비밀번호를 여러 번 틀려서 10분 동안 잠겼어요. 잠시 후 다시 시도해 주세요.' };

  const saved = String(settings['관리자비밀번호'] || '').trim();
  if (!saved) {
    ensureSetting_('관리자비밀번호', '관리자 페이지(admin.html) 비밀번호 — 직접 정해서 적어 주세요');
    return { ok: false, code: 'no_password', message: '구글 시트 설정 탭의 "관리자비밀번호" 칸에 비밀번호를 먼저 적어 주세요.' };
  }
  if (String(req.pw || '') !== saved) {
    cache.put('adminFails', String(fails + 1), 600);
    return { ok: false, code: 'bad_password', message: '비밀번호가 맞지 않아요.' };
  }
  cache.remove('adminFails');

  const action = String(req.action);
  if (action === 'admin_list') return adminList_();

  lock.waitLock(20000);
  const date = normDate_(String(req.date || ''), 'Asia/Seoul');
  const cls = String(req.cls || '').trim();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) return { ok: false, message: '날짜가 올바르지 않아요.' };
  if (cls && !listClasses_().some(c => c.name === cls)) return { ok: false, message: '알 수 없는 수업이에요: ' + cls };

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tz = ss.getSpreadsheetTimeZone();
  const off = ss.getSheetByName(SHEET_OFF);

  if (action === 'admin_close') {
    const rows = off.getDataRange().getValues().slice(1);
    const dup = rows.some(r => r[0] !== '' && normDate_(r[0], tz) === date && String(r[1] || '').trim() === cls);
    if (!dup) off.appendRow(["'" + date, cls, '관리자 페이지에서 마감']);
  } else if (action === 'admin_open') {
    const values = off.getDataRange().getValues();
    for (let i = values.length - 1; i >= 1; i--) {         // 아래에서부터 지워야 줄 번호가 안 꼬여요
      const r = values[i];
      if (r[0] !== '' && normDate_(r[0], tz) === date && String(r[1] || '').trim() === cls) off.deleteRow(i + 1);
    }
  } else if (action === 'admin_cancel') {
    const row = Number(req.row);
    const sheet = ss.getSheetByName(SHEET_BOOKINGS);
    if (!(row >= 2 && row <= sheet.getLastRow())) return { ok: false, message: '예약을 찾지 못했어요. 새로고침해 주세요.' };
    const r = sheet.getRange(row, 1, 1, 7).getValues()[0];
    // 줄이 바뀌었을 수 있으니 날짜·수업·연락처가 맞는지 한 번 더 확인
    if (normDate_(r[1], tz) !== date || String(r[2]).trim() !== cls || normPhone_(r[4]) !== normPhone_(req.phone)) {
      return { ok: false, message: '예약 정보가 바뀌었어요. 새로고침 후 다시 시도해 주세요.' };
    }
    sheet.getRange(row, 7).setValue(CANCELLED);
    SpreadsheetApp.flush();
    try {
      const c = listClasses_().find(x => x.name === cls);
      if (c) upsertEvent_(date, c, activeBookings_().filter(b => b.date === date && b.cls === cls), settings);
    } catch (calErr) { console.error('캘린더 갱신 실패', calErr); }
  } else {
    return { ok: false, message: '알 수 없는 요청이에요.' };
  }
  SpreadsheetApp.flush();
  return adminList_();
}

function adminList_() {
  const sched = buildSchedule_();
  const first = sched.days[0].date, last = sched.days[sched.days.length - 1].date;
  const bookings = activeBookings_()
    .filter(b => b.date >= first && b.date <= last)
    .map(b => ({ row: b.row, date: b.date, cls: b.cls, name: b.name, phone: b.phone, rental: b.rental }));
  return Object.assign({ ok: true, bookings: bookings }, sched);
}

function ensureSetting_(key, desc) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SETTINGS);
  const has = sh.getDataRange().getValues().some(r => String(r[0]).trim() === key);
  if (!has) sh.appendRow([key, '', desc]);
}

/* ───────────── 3. 예약 가능 현황 계산 ───────────── */

function buildSchedule_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tz = ss.getSpreadsheetTimeZone();
  const settings = getSettings_();
  const classes = listClasses_();
  const horizon = Number(settings['예약가능일수']) || 60;
  const cutoffH = Number(settings['예약마감(시간전)']);
  const cutoffMs = (isNaN(cutoffH) ? 12 : cutoffH) * 3600 * 1000;
  const globalConflict = String(settings['겹침마감'] || '예').trim() !== '아니오';

  // 날짜+수업별 예약 인원
  const counts = {};
  activeBookings_().forEach(b => {
    const k = b.date + '|' + b.cls;
    counts[k] = (counts[k] || 0) + 1;
  });

  // 휴무: 수업 칸이 비면 하루 전체, 아니면 그 수업만
  const offRows = ss.getSheetByName(SHEET_OFF).getDataRange().getValues().slice(1);
  const offDay = {}, offClass = {}, offClassList = {};
  offRows.forEach(r => {
    if (r[0] === '') return;
    const d = normDate_(r[0], tz);
    const c = String(r[1] || '').trim();
    if (c) { offClass[d + '|' + c] = true; (offClassList[d] = offClassList[d] || []).push(c); }
    else offDay[d] = true;
  });

  const now = new Date();
  const todayKst = Utilities.formatDate(now, 'Asia/Seoul', 'yyyy-MM-dd');
  const holidays = getHolidays_(todayKst, addDays_(todayKst, horizon + 1), settings);
  const days = [];
  for (let i = 0; i <= horizon; i++) {
    const date = addDays_(todayKst, i);
    const weekday = WEEK[toKst_(date, '12:00').getUTCDay()];
    const slots = {};
    let anyOpen = false;

    classes.forEach(c => {
      const booked = counts[date + '|' + c.name] || 0;
      let reason = '';
      if (c.days.indexOf(weekday) < 0) reason = weekday + '요일 휴무';
      else if (holidays[date] && !c.holidayOpen) reason = '공휴일 휴무';
      else if (offDay[date] || offClass[date + '|' + c.name]) reason = '휴무';
      else if (toKst_(date, c.start).getTime() - now.getTime() < cutoffMs) reason = '예약 마감';
      else if (booked >= c.capacity) reason = '정원 마감';
      else if (globalConflict && c.conflicts.some(o => (counts[date + '|' + o] || 0) > 0)) reason = '겹치는 수업 진행';
      slots[c.name] = { booked: booked, closed: !!reason, reason: reason };
      if (!reason) anyOpen = true;
    });

    days.push({
      date: date, weekday: weekday, open: anyOpen, slots: slots, holiday: holidays[date] || '',
      offDay: !!offDay[date], offClasses: offClassList[date] || [],
    });
  }

  return {
    classes: classes.map(c => ({
      name: c.name, start: c.start, end: c.end, capacity: c.capacity,
      conflicts: c.conflicts, place: c.place,
    })),
    days: days,
    rentalItems: RENTAL_ITEMS,
  };
}

function listClasses_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tz = ss.getSpreadsheetTimeZone();
  return ss.getSheetByName(SHEET_CLASSES).getDataRange().getValues().slice(1)
    .filter(r => String(r[0]).trim() !== '' && r[1] !== '')
    .map(r => ({
      name: String(r[0]).trim(),
      start: normTime_(r[1], tz),
      end: r[2] === '' ? '' : normTime_(r[2], tz),
      capacity: Number(r[3]) || 4,
      conflicts: String(r[4] || '').split(/[,，/]/).map(s => s.trim()).filter(Boolean),
      days: String(r[5] || '월화수목금토일'),
      place: String(r[6] || ''),
      holidayOpen: String(r[7] || '아니오').trim() === '예',
    }))
    .sort((a, b) => a.start < b.start ? -1 : 1);
}

function activeBookings_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tz = ss.getSpreadsheetTimeZone();
  return ss.getSheetByName(SHEET_BOOKINGS).getDataRange().getValues().slice(1)
    .map((r, i) => { r.rowNum = i + 2; return r; })
    .filter(r => r[1] !== '' && String(r[6]).trim() !== CANCELLED)
    .map(r => ({
      row: r.rowNum,
      date: normDate_(r[1], tz),
      cls: String(r[2]).trim(),
      name: String(r[3]).trim(),
      phone: normPhone_(r[4]) || String(r[4]).trim(),
      rental: String(r[5] || ''),
    }));
}

function getSettings_() {
  const rows = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SETTINGS)
    .getDataRange().getValues().slice(1);
  const map = {};
  rows.forEach(r => { map[String(r[0]).trim()] = r[1]; });
  return map;
}

/* 공휴일 목록 { '2026-10-09': '한글날', ... }
   구글의 '대한민국의 휴일' 캘린더에서 가져와요. 어버이날 같은 '기념일'은 빼고 쉬는 날만.
   6시간 동안 기억해 두어서 페이지가 빨리 열려요. */
function getHolidays_(fromDate, toDate, settings) {
  const cache = CacheService.getScriptCache();
  const key = 'holidays|' + fromDate + '|' + toDate;
  const hit = cache.get(key);
  if (hit) return JSON.parse(hit);

  const map = {};
  try {
    const id = String(settings['공휴일캘린더ID'] || HOLIDAY_CAL_DEFAULT).trim();
    let cal = CalendarApp.getCalendarById(id);
    if (!cal) cal = CalendarApp.subscribeToCalendar(id, { hidden: true, selected: false });
    cal.getEvents(toKst_(fromDate, '00:00'), toKst_(toDate, '00:00')).forEach(ev => {
      const desc = String(ev.getDescription() || '');
      if (/기념일|Observance/i.test(desc)) return;           // 쉬지 않는 기념일은 제외
      const allDay = ev.isAllDayEvent();
      const tzUse = allDay ? Session.getScriptTimeZone() : 'Asia/Seoul';
      let d = Utilities.formatDate(allDay ? ev.getAllDayStartDate() : ev.getStartTime(), tzUse, 'yyyy-MM-dd');
      const endD = Utilities.formatDate(allDay ? ev.getAllDayEndDate() : ev.getEndTime(), tzUse, 'yyyy-MM-dd');
      do { map[d] = ev.getTitle(); d = addDays_(d, 1); } while (d < endD);
    });
    cache.put(key, JSON.stringify(map), 6 * 3600);
  } catch (err) {
    console.error('공휴일을 불러오지 못했어요 (휴무 탭에 직접 적어 주세요)', err);
  }
  return map;
}

/* ───────────── 4. 작은 도우미들 ───────────── */

/* 연락처를 010-1234-5678 모양으로 통일 */
function normPhone_(v) {
  const d = String(v || '').replace(/\D/g, '');
  if (!/^01\d{8,9}$/.test(d)) return '';
  return d.length === 11
    ? d.slice(0, 3) + '-' + d.slice(3, 7) + '-' + d.slice(7)
    : d.slice(0, 3) + '-' + d.slice(3, 6) + '-' + d.slice(6);
}

/* 렌탈 장비: '모두'는 4가지 전부, '없음'은 대여 안 함 */
function normRental_(v) {
  const list = (Array.isArray(v) ? v : [v]).map(s => String(s).trim());
  if (list.indexOf('모두') >= 0) return RENTAL_ITEMS.join(', ');
  const picked = RENTAL_ITEMS.filter(i => list.indexOf(i) >= 0);
  if (picked.length) return picked.join(', ');
  if (list.indexOf('없음') >= 0) return '없음';
  return '';
}

/* 날짜를 2026-10-05 모양으로 통일 (2026.10.5, 2026/10/5 도 OK) */
function normDate_(v, tz) {
  if (v instanceof Date) return Utilities.formatDate(v, tz, 'yyyy-MM-dd');
  const p = String(v).trim().split(/[-./\s]+/).filter(Boolean);
  if (p.length < 3) return String(v).trim();
  return p[0] + '-' + ('0' + p[1]).slice(-2) + '-' + ('0' + p[2]).slice(-2);
}

/* 시간을 09:30 모양으로 통일 (9:30, 9시30분 도 OK) */
function normTime_(v, tz) {
  if (v instanceof Date) return Utilities.formatDate(v, tz, 'HH:mm');
  const m = String(v).trim().match(/(\d{1,2})\D*(\d{2})?/);
  if (!m) return String(v).trim();
  return ('0' + m[1]).slice(-2) + ':' + (m[2] || '00');
}

/* 한국 시간 기준 Date 만들기 */
function toKst_(date, time) {
  return new Date(date + 'T' + (time || '00:00') + ':00+09:00');
}

function addDays_(date, n) {
  const d = toKst_(date, '12:00');
  d.setUTCDate(d.getUTCDate() + n);
  return Utilities.formatDate(d, 'Asia/Seoul', 'yyyy-MM-dd');
}

/* ───────────── 5. 구글 캘린더 ───────────── */

function getCalendar_(settings) {
  const id = String(settings['캘린더ID'] || 'primary').trim();
  return id === 'primary' ? CalendarApp.getDefaultCalendar() : CalendarApp.getCalendarById(id);
}

/* 한 수업의 캘린더 일정을 만들거나 명단을 고쳐요 */
function upsertEvent_(date, cls, people, settings) {
  const cal = getCalendar_(settings);
  const tag = date + '|' + cls.name;
  const start = toKst_(date, cls.start);
  const end = cls.end ? toKst_(date, cls.end) : new Date(start.getTime() + 2 * 3600 * 1000);
  const existing = cal.getEvents(start, end).filter(ev => ev.getTag('bluedogSlot') === tag);

  if (people.length === 0) { existing.forEach(ev => ev.deleteEvent()); return; }

  // 렌탈 장비 합계 (준비물 챙기기용)
  const totals = {};
  people.forEach(p => RENTAL_ITEMS.forEach(i => { if (p.rental.indexOf(i) >= 0) totals[i] = (totals[i] || 0) + 1; }));
  const rentalLine = RENTAL_ITEMS.filter(i => totals[i]).map(i => i + ' ' + totals[i]).join(' · ') || '없음';

  const title = '🤿 ' + cls.name + ' ' + people.length + '/' + cls.capacity + '명';
  const desc = '[블루독 예약 명단]\n' +
    people.map((p, i) => (i + 1) + '. ' + p.name + ' ' + p.phone + ' / 렌탈: ' + p.rental).join('\n') +
    '\n\n[렌탈 준비] ' + rentalLine;

  if (existing.length) {
    existing[0].setTitle(title).setDescription(desc);
    existing.slice(1).forEach(ev => ev.deleteEvent());
  } else {
    const ev = cal.createEvent(title, start, end, { location: cls.place, description: desc });
    ev.setTag('bluedogSlot', tag);
  }
}

/* 취소·수정 후 메뉴에서 실행: 앞으로 있을 모든 수업의 캘린더를 다시 맞춰요 */
function syncAllCalendar() {
  const settings = getSettings_();
  const classes = listClasses_();
  const bookings = activeBookings_();
  const todayKst = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd');
  const horizon = Number(settings['예약가능일수']) || 60;

  // 예약이 있었던 날짜만 확인 (빠르게)
  const sheetDates = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_BOOKINGS)
    .getDataRange().getValues().slice(1)
    .map(r => r[1] === '' ? '' : normDate_(r[1], SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone()));
  const dates = Array.from(new Set(sheetDates))
    .filter(d => d && d >= todayKst && d <= addDays_(todayKst, horizon));

  dates.forEach(date => classes.forEach(cls => {
    upsertEvent_(date, cls, bookings.filter(b => b.date === date && b.cls === cls.name), settings);
  }));
}
